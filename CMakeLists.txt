cmake_minimum_required(VERSION 3.24)
project(windows-storage-health VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows is required")
endif()
option(WSH_ANALYZE "Enable MSVC static analysis" OFF)
option(WSH_ASAN "Enable address sanitizer" OFF)
add_library(wsh_core src/model.cpp src/cli.cpp src/correlation.cpp src/analysis.cpp src/json.cpp src/human.cpp src/storage.cpp src/storage_common.cpp src/properties.cpp src/health.cpp src/disks.cpp src/volumes.cpp)
target_include_directories(wsh_core PUBLIC include)
target_compile_features(wsh_core PUBLIC cxx_std_17)
target_compile_definitions(wsh_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(wsh_core PUBLIC setupapi cfgmgr32)
add_executable(windows-storage-health src/main.cpp)
target_link_libraries(windows-storage-health PRIVATE wsh_core)
if(MSVC)
  foreach(t wsh_core windows-storage-health)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    if(WSH_ANALYZE)
      target_compile_options(${t} PRIVATE /analyze)
    endif()
    if(WSH_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
      target_link_options(${t} PRIVATE /INCREMENTAL:NO)
    endif()
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(wsh_tests tests/tests.cpp)
  target_link_libraries(wsh_tests PRIVATE wsh_core)
  if(MSVC)
    target_compile_options(wsh_tests PRIVATE /W4 /utf-8)
    if(WSH_ASAN)
      target_compile_options(wsh_tests PRIVATE /fsanitize=address /Zi)
      target_link_options(wsh_tests PRIVATE /INCREMENTAL:NO)
    endif()
  endif()
  add_test(NAME synthetic COMMAND wsh_tests)
  add_test(NAME json_parse COMMAND ${CMAKE_COMMAND} -DTEST_PROGRAM=$<TARGET_FILE:wsh_tests> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/json.cmake)
  add_test(NAME help COMMAND windows-storage-health --help)
  add_test(NAME version COMMAND windows-storage-health --version)
endif()
install(TARGETS windows-storage-health RUNTIME DESTINATION bin)
